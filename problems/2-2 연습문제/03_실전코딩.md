
## 정답

## 정답 코드

```python
import os
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI

GMS_BASE_URL = "https://gms.ssafy.io/gmsapi/api.openai.com/v1/"

def ask_gms(question: str) -> str:
    load_dotenv()
    api_key = os.getenv("GMS_KEY")
    if not api_key:
        raise RuntimeError("GMS_KEY 환경변수가 필요합니다.")

    llm = ChatOpenAI(
        model="gpt-5.4-mini",
        api_key=api_key,
        base_url=GMS_BASE_URL,
    )
    return llm.invoke(question).content


print(ask_gms("생성형 AI를 한 문장으로 설명하세요."))
```

## 핵심 해설

- `.env` 로드 → `os.getenv` → 키 검증 → `ChatOpenAI` 생성 → `invoke` → `.content` 반환 순서다.
- 입력 `question: str`은 `invoke`로 전달되고, 반환된 메시지 객체의 `content: str`이 함수 출력이 된다.
- 모델 변경은 `model`, 서버 변경은 `base_url`, 인증 변수 변경은 `os.getenv` 블록을 수정한다.

```python
classify_sentiment("배송이 빨라 만족합니다.", llm, "few")
# 예상 형식: "긍정"
```

```python
def classify_sentiment(text, llm, mode="zero"):
    if mode == "zero":
        prompt = (
            "다음 문장의 감성을 '긍정' 또는 '부정' 중 하나로만 답하세요.\n"
            f"문장: {text}\n감성:"
        )
    elif mode == "few":
        prompt = f"""다음 문장의 감성을 '긍정' 또는 '부정' 중 하나로만 답하세요.

문장: 정말 최고의 서비스였습니다.
감성: 긍정

문장: 배송이 너무 늦어서 실망했어요.
감성: 부정

문장: {text}
감성:"""
    else:
        raise ValueError("mode는 'zero' 또는 'few'여야 합니다.")

    return llm.invoke(prompt).content.strip()
```

## 핵심 해설

- mode 분기 → 방식별 프롬프트 조립 → 공통 `invoke` → `.content.strip()` 순서다.
- Few-shot 예시는 입력과 출력의 대응 및 허용 레이블을 다음 입력으로 전달한다.
- 중립을 추가하려면 허용 레이블 문구와 Few-shot 예시를 함께 수정해야 한다. 분류 작업 자체가 바뀌면 프롬프트 구성 블록만 교체하고 호출 블록은 재사용할 수 있다.

## 정답

## 정답 코드

```python
import os
from dotenv import load_dotenv
from openai import OpenAI

load_dotenv()

def complete(prompt: str, max_output_tokens: int = 500) -> str:
    client = OpenAI(
        api_key=os.getenv("GMS_KEY"),
        base_url="https://gms.ssafy.io/gmsapi/api.openai.com/v1/",
    )
    response = client.chat.completions.create(
        model="gpt-5.4-mini",
        messages=[{"role": "user", "content": prompt}],
        temperature=1.0,
        max_completion_tokens=max_output_tokens,
    )
    return response.choices[0].message.content
```

## 핵심 해설

- 환경 설정 → 클라이언트 생성 → `messages` 구성 → `chat.completions.create` → 첫 choice 본문 반환 순서다.
- `prompt`는 `messages[0]["content"]`로, 함수의 길이 인자는 `max_completion_tokens`로 전달된다.
- 시스템 메시지나 대화 기록이 필요하면 `messages` 리스트에 `{"role": ..., "content": ...}` 원소를 호출 순서대로 추가한다.

## 정답

## 정답 코드

```python
import json

def generate_reviews(product: str, llm) -> list[dict]:
    review_properties = {
        "customer_name": {"type": "string", "description": "고객 이름"},
        "rating": {"type": "integer", "description": "별점 (1-5)"},
        "category": {"type": "string", "description": "긍정/부정/중립"},
        "text": {"type": "string", "description": "리뷰 본문"},
    }
    response_format = {
        "type": "json_schema",
        "json_schema": {
            "name": "customer_reviews",
            "strict": True,
            "schema": {
                "type": "object",
                "properties": {
                    "reviews": {
                        "type": "array",
                        "items": {
                            "type": "object",
                            "properties": review_properties,
                            "required": list(review_properties),
                            "additionalProperties": False,
                        },
                    }
                },
                "required": ["reviews"],
                "additionalProperties": False,
            },
        },
    }

    prompt = (
        f"{product} 고객 리뷰 3개를 생성하세요. "
        "긍정, 부정, 중립을 각각 하나씩 포함하세요."
    )
    response = llm.invoke(prompt, response_format=response_format)
    return json.loads(response.content)["reviews"]
```

## 핵심 해설

- 필드 명세 → 리뷰 객체 스키마 → 배열·최상위 객체 스키마 → `response_format` 래핑 → 호출 → 파싱 순서다.
- `list(review_properties)`는 딕셔너리 키 리스트가 되어 네 필드를 모두 `required`로 전달한다.
- 필드를 바꾸려면 `review_properties`를 수정하면 `properties`와 `required`가 함께 반영된다. 리뷰 개수와 감성 구성은 프롬프트 블록을 수정한다.
- `response.content`는 JSON 문자열이며 `json.loads(...)["reviews"]`가 최종 리스트 출력이다.

```python
import json

def build_dataset(personas, products, llm, output_path):
    results = []
    for persona in personas:
        for product in products:
            prompt = f"{persona}의 관점에서 {product} 리뷰를 작성하세요."
            review = llm.invoke(prompt).content
            results.append({
                "persona": persona,
                "product": product,
                "review": review,
            })

    with open(output_path, "w", encoding="utf-8") as file:
        json.dump(results, file, ensure_ascii=False, indent=2)
    return results


personas = ["20대 대학생", "40대 주부", "60대 은퇴자"]
products = ["노트북", "운동화"]
dataset = build_dataset(personas, products, llm, "synthetic_reviews.json")
print(len(dataset))
```

# 문제 6. LLM-as-a-Judge 품질 필터 파이프라인

- LLM은 `specificity`, `naturalness`, `usefulness`, `total`, `reason`을 가진 JSON으로 답해야 한다.
- `total`은 세 항목의 평균이다.

- `scores["total"] >= threshold`인 항목만 반환한다.
- 함수 반환값은 `(passed, pass_rate)` 튜플이다.
- 입력 리뷰가 비어 있으면 `([], 0.0)`을 반환한다.

## 제공 코드 또는 스켈레톤

```python
def filter_reviews(reviews, llm, threshold=3.5):
    pass
```

## 예상 입력·출력

```python
passed, rate = filter_reviews(
    ["좋아요.", "배터리가 8시간 지속되고 무게가 200g이라 휴대하기 좋습니다."],
    llm,
)
# passed: 합격한 원문과 점수를 묶은 딕셔너리 리스트
# rate: 0.0~100.0 사이의 통과율
```

## 정답

## 정답 코드

```python
import json

EVAL_PROMPT = """다음 리뷰의 품질을 1-5점으로 평가하세요.

평가 기준:
1. 구체성: 제품의 구체적인 특징을 언급하는가?
2. 자연스러움: 실제 고객이 쓴 것처럼 자연스러운가?
3. 유용성: 다른 구매자에게 도움이 되는가?

리뷰: {review_text}

JSON 형식으로 답하세요. total은 세 점수의 평균입니다.
{{"specificity": 1, "naturalness": 1, "usefulness": 1,
  "total": 1, "reason": "평가 근거"}}"""

def evaluate_review(review_text, llm):
    prompt = EVAL_PROMPT.format(review_text=review_text)
    response = llm.invoke(prompt)
    return json.loads(response.content)

def filter_reviews(reviews, llm, threshold=3.5):
    if not reviews:
        return [], 0.0

    evaluated = []
    for review in reviews:
        scores = evaluate_review(review, llm)
        evaluated.append({"review": review, "scores": scores})

    passed = [
        item for item in evaluated
        if item["scores"]["total"] >= threshold
    ]
    pass_rate = len(passed) / len(evaluated) * 100
    return passed, pass_rate
```

## 핵심 해설

- 입력 검증 → 평가 프롬프트 채움 → 호출 → JSON 파싱 → 원문과 점수 결합 → `total` 필터 → 통과율 계산 순서다.
- 리뷰 문자열은 `format`으로 프롬프트에 전달되고, 점수 문자열은 `json.loads`로 딕셔너리가 된 뒤 원문과 결합된다.
- JSON 예시의 중괄호는 `str.format`에서 보존되도록 `{{`, `}}`로 작성한다.
- 평가 기준을 바꾸면 프롬프트의 기준·출력 필드를, 합격 강도를 바꾸면 `threshold`를 수정한다. 평가 JSON의 중첩 위치가 바뀌면 필터의 `item["scores"]["total"]` 접근도 바꿔야 한다.

[문제로 돌아가기](#question6)
